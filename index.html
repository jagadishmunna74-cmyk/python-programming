# Function to display ASCII character table
def display_ascii_table():
    start_char = 33  # ASCII value for '!'
    end_char = 126   # ASCII value for '~'
    chars_per_line = 5

    print(f"{'Char':^5} {'Dec':^5} {'Hex':^5} " * chars_per_line)
    print('-' * 21 * chars_per_line)

    for i in range(start_char, end_char + 1):
        char = chr(i)
        dec_val = i
        hex_val = hex(i)

        print(f"{char:^5} {dec_val:^5} {hex_val:^5}", end=' ')

        if (i - start_char + 1) % chars_per_line == 0:
            print()

    if (end_char - start_char + 1) % chars_per_line != 0:
        print()

# Call the function to display the table
display_ascii_table()

"""

# Function to calculate the Fibonacci number using recursion
def fibonacci(n):
    if n == 1 or n == 2:
        return 1
    else:
        return fibonacci(n-1) + fibonacci(n-2)

# Function to print Fibonacci sequence up to 'n' numbers
def print_fibonacci_sequence(n):
    for i in range(1, n+1):
        print(fibonacci(i), end=" ")

# Example usage
n = int(input("Enter the number of terms: "))
print("Fibonacci Sequence:")
print_fibonacci_sequence(n)

"""Practical 8(D) Write a program that defines a function to determine whether input number n is prime or not. A positive whole number n > 2 is prime, if no
number between 2 and 𝑛 (inclusive) evenly divides n. If n is not prime, the program should quit as soon as it finds a value that evenly divides n.

"""

def is_prime(n):
    # If the number is less than 2, it's not prime
    if n <= 1:
        return False

    # Check divisibility starting from 2 to sqrt(n)
    for i in range(2, int(n**0.5) + 1):
        if n % i == 0:
            return False  # If n is divisible by i, it's not prime

    # If no divisors found, the number is prime
    return True

# Example usage
n = int(input("Enter a number: "))
if is_prime(n):
    print(f"{n} is a prime number.")
else:
    print(f"{n} is not a prime number.")
